<script>
// ===== 背景切换逻辑 =====
const bgBox = document.getElementById("bgBox");
const bgButtons = document.querySelectorAll(".bg-btn");
// 默认第一张
bgBox.style.backgroundImage = `url("https://picsum.photos/id/1015/1920/1080")`;

bgButtons.forEach(btn => {
  btn.style.backgroundImage = `url(${btn.dataset.bg})`;
  btn.onclick = () => {
    bgBox.style.backgroundImage = `url(${btn.dataset.bg})`;
  }
})

// ===== 竖日期生成 =====
const totalDays = 31;
const dateWrap = document.getElementById("dateList");

for (let i = 1; i <= totalDays; i++) {
  const item = document.createElement("div");
  item.className = "date-item";
  if (i === 1) item.classList.add("active");
  item.innerText = i;
  item.dataset.target = i;

  item.onclick = () => {
    document.querySelectorAll(".date-item").forEach(d => d.classList.remove("active"));
    item.classList.add("active");
    document.querySelectorAll(".day-panel").forEach(p => p.classList.remove("show"));
    const target = document.querySelector(`.day-panel[data-day="${i}"]`);
    if (target) target.classList.add("show");
  };
  dateWrap.appendChild(item);
}

// ===== 预加载淡出逻辑 =====
var loader = document.getElementById("preloader");

function hideLoader() {
    if (!loader) return;
    fadeOut();
}

function fadeOut() {
    var op = 1;
    var fade = setInterval(function () {
        op -= 0.1;
        if (op <= 0) {
            clearInterval(fade);
            loader.style.display = "none";
        } else {
            loader.style.opacity = op;
        }
    }, 40);
}

window.addEventListener("load", hideLoader);
setTimeout(hideLoader, 8000);
</script>
